cmake_minimum_required(VERSION 3.22)
project(agent_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS agent_compile=2 agent_link=1)
set(CMAKE_JOB_POOL_COMPILE agent_compile)
set(CMAKE_JOB_POOL_LINK agent_link)
set(CMAKE_C_FLAGS_DEBUG "-O2" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O2" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp-b5124 llama)
if(ANDROID)
  add_library(agent_native SHARED jni.cpp)
  target_link_libraries(agent_native PRIVATE llama log)
else()
  add_executable(agent-smoke smoke.cpp)
  target_link_libraries(agent-smoke PRIVATE llama)
endif()
